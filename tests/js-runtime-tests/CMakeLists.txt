foreach(_axjs_test IN LISTS _axjs_test_targets)
    target_link_libraries(${_axjs_test} PRIVATE axjs)
    target_compile_features(${_axjs_test} PRIVATE cxx_std_23)
    if(ANDROID)
        target_link_libraries(${_axjs_test} PRIVATE cpp_android_spec)
    endif()
    if(APPLE)
        set_target_properties(${_axjs_test} PROPERTIES BUILD_WITH_INSTALL_RPATH FALSE SKIP_BUILD_RPATH FALSE)
        if(TARGET OpenAL)
            get_target_property(_axjs_openal_framework OpenAL FRAMEWORK)
            if(_axjs_openal_framework)
                if(XCODE)
                    # OpenAL inherits Xcode's products directory. Link the framework
                    # by its Xcode product reference, as Axmol does for applications.
                    set_target_properties(${_axjs_test} PROPERTIES
                        XCODE_LINK_BUILD_PHASE_MODE KNOWN_LOCATION
                        XCODE_ATTRIBUTE_LD_RUNPATH_SEARCH_PATHS
                            "$(inherited) \"$(SYMROOT)/$(CONFIGURATION)$(EFFECTIVE_PLATFORM_NAME)\"")
                else()
                    set_property(TARGET ${_axjs_test} APPEND PROPERTY BUILD_RPATH "$<TARGET_BUNDLE_DIR:OpenAL>/..")
                endif()
            endif()
        endif()
        target_link_libraries(${_axjs_test} PRIVATE "-framework Cocoa" "-framework Metal" "-framework QuartzCore")
    endif()
    if(_axjs_test STREQUAL "js-shader-archive-rejection-tests")
        add_test(NAME ${_axjs_test} COMMAND ${_axjs_test} "${AXSLCC_OUT_DIR}"
            "${CMAKE_CURRENT_BINARY_DIR}/shader-rejection-fixtures")
        set_tests_properties(${_axjs_test} PROPERTIES SKIP_RETURN_CODE 77)
    elseif(_axjs_test STREQUAL "js-legacy-bytecode-tests")
        add_test(NAME ${_axjs_test} COMMAND ${_axjs_test} "${_axjs_legacy_fixture}")
    elseif(_axjs_test STREQUAL "js-legacy-source-tests" OR
           _axjs_test STREQUAL "js-legacy-action-duration-tests" OR
           _axjs_test STREQUAL "js-physics3d-kinematic-tests" OR
           _axjs_test STREQUAL "js-scheduler-frame-hook-tests")
        add_test(NAME ${_axjs_test} COMMAND ${_axjs_test} "${_axjs_test_source_root}")
    elseif(_axjs_test STREQUAL "js-tiff-tests")
        add_test(NAME ${_axjs_test} COMMAND ${_axjs_test} "${_axjs_tiff_fixtures}")
    else()
        add_test(NAME ${_axjs_test} COMMAND ${_axjs_test})
    endif()
    set_tests_properties(${_axjs_test} PROPERTIES TIMEOUT 90)
endforeach()
